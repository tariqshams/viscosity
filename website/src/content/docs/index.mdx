```js
// you write:
const ps = [];
for (let i = 0; i < n; i++) ps.push({ x: i, vx: 1 });
for (const p of ps) p.x += p.vx * dt;

// Viscosity emits (abridged):
const ps = new __Vsc_ps();            // one Float64Array per field
for (let i = 0; i < n; i++) ps.push(i, 1);
for (let p = 0; p < ps.length; p++) ps.x[p] += ps.vx[p] * dt;
```
